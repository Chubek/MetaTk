cmake_minimum_required(VERSION 3.16)
project(GPPX VERSION 1.0.1 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(GPPX_BUILD_TESTS "Build and run unit/integration tests" ON)
option(GPPX_BUILD_WEBBER "Build sample Webber shared plugin" ON)
option(GPPX_BUILD_VMSPECS "Build the VM reference catalogue" ON)
option(GPPX_BUILD_EXAMPLES "Build the mylang example" ON)
option(GPPX_SANITIZERS "Enable address/undefined sanitizers (GCC/Clang)" OFF)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(GPPX_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
add_library(gppx SHARED GPPX.cpp GPPX.c)
add_library(GPPX::gppx ALIAS gppx)
set_target_properties(gppx PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(gppx PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(gppx PRIVATE GPPX_BUILD_SHARED)
if(CMAKE_DL_LIBS)
  target_link_libraries(gppx PRIVATE ${CMAKE_DL_LIBS})
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(gppx PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(gppx_cli driver/gppx.cpp driver/config.cpp driver/loader.cpp driver/repl.cpp)
set_target_properties(gppx_cli PROPERTIES OUTPUT_NAME gppx)
target_include_directories(gppx_cli PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(gppx_cli PRIVATE GPPX::gppx)
if(APPLE)
 set_target_properties(gppx_cli PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
elseif(UNIX)
 set_target_properties(gppx_cli PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
endif()
if(GPPX_BUILD_WEBBER)
 add_library(gppx_webber MODULE plugins/webber/Assembler.cpp)
 target_include_directories(gppx_webber PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 set_target_properties(gppx_webber PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
endif()
if(GPPX_BUILD_EXAMPLES)
 add_executable(gppx_mylang examples/mylang.cpp)
 target_link_libraries(gppx_mylang PRIVATE GPPX::gppx)
endif()
if(GPPX_BUILD_TESTS)
 enable_testing()
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(UNIX AND Python3_Interpreter_FOUND)
  find_program(GPPX_VIM_EXECUTABLE NAMES vim)
  if(GPPX_VIM_EXECUTABLE)
   add_test(NAME gppx_vim_addon COMMAND "${Python3_EXECUTABLE}"
    "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_addons.py" "${GPPX_VIM_EXECUTABLE}")
  endif()
 endif()
 if(GPPX_BUILD_EXAMPLES)
  add_test(NAME gppx_mylang_demo COMMAND gppx_mylang)
  set_tests_properties(gppx_mylang_demo PROPERTIES PASS_REGULAR_EXPRESSION "greet\\(world\\);")
 endif()
 add_library(gppx_shout MODULE examples/c_plugin/shout.c)
 target_include_directories(gppx_shout PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
 set_target_properties(gppx_shout PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/plugins")
 add_test(NAME gppx_c_plugin COMMAND $<TARGET_FILE:gppx_cli> --plugin $<TARGET_FILE:gppx_shout> -e "#shout hello")
 set_tests_properties(gppx_c_plugin PROPERTIES PASS_REGULAR_EXPRESSION "HELLO!")
 add_executable(gppx_tests tests/test.cpp)
 target_include_directories(gppx_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_link_libraries(gppx_tests PRIVATE GPPX::gppx)
 target_compile_definitions(gppx_tests PRIVATE GPPX_TEST_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
 if(GPPX_BUILD_WEBBER)
  target_compile_definitions(gppx_tests PRIVATE GPPX_WEBBER_PATH="$<TARGET_FILE:gppx_webber>")
  add_dependencies(gppx_tests gppx_webber)
 endif()
 add_executable(gppx_extension_tests tests/test_extensions.c)
 target_link_libraries(gppx_extension_tests PRIVATE GPPX::gppx)
 target_include_directories(gppx_extension_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
 target_compile_definitions(gppx_extension_tests PRIVATE GPPX_TEST_DATA_PATH="${CMAKE_CURRENT_BINARY_DIR}/gppx_extension_data.tmp")
 add_test(NAME gppx_extension_c_api COMMAND gppx_extension_tests)
 add_test(NAME gppx_core COMMAND gppx_tests)
 add_test(NAME gppx_cli_smoke COMMAND $<TARGET_FILE:gppx_cli> --version)
 add_test(NAME gppx_cli_config COMMAND $<TARGET_FILE:gppx_cli> --config ${CMAKE_CURRENT_SOURCE_DIR}/driver/config.yaml -e "#include \"plain.gppx\"")
 set_tests_properties(gppx_cli_config PROPERTIES PASS_REGULAR_EXPRESSION "PLAIN")
 if(GPPX_BUILD_WEBBER)
 add_test(NAME gppx_cli_plugin_directory COMMAND $<TARGET_FILE:gppx_cli> --plugin-dir ${CMAKE_CURRENT_BINARY_DIR}/plugins -e "#wbr_upper installed")
 set_tests_properties(gppx_cli_plugin_directory PROPERTIES PASS_REGULAR_EXPRESSION "INSTALLED")
 endif()

endif()
install(TARGETS gppx gppx_cli EXPORT GPPXTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/GPPX.h include/GPPX.hpp include/GPPX-all.h include/GPPX-Directive.h include/GPPX-Macro.h include/GPPX-Syntax.h include/GPPX-Reader.h include/GPPX-Sink.h include/GPPX-Embed.h include/GPPX-Rewriter.h include/GPPX-PMatch.h include/GPPX-Style.h include/GPPX-Layout.h include/GPPX-UTF8Input.h include/GPPX-MetaSym.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT GPPXTargets NAMESPACE GPPX:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/GPPX)
if(GPPX_BUILD_WEBBER)
 install(TARGETS gppx_webber LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/gppx/plugins)
endif()

configure_package_config_file(cmake/GPPXConfig.cmake.in
  "${CMAKE_CURRENT_BINARY_DIR}/GPPXConfig.cmake"
  INSTALL_DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GPPX")
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/GPPXConfigVersion.cmake"
  VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/GPPXConfig.cmake"
  "${CMAKE_CURRENT_BINARY_DIR}/GPPXConfigVersion.cmake"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/GPPX")

if(GPPX_BUILD_VMSPECS)
 add_subdirectory(VMSpec)
endif()
