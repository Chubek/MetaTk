option(METATEX_ENABLE_LUA "Build Lua 5.3+ native-engine binding" ON)
if(NOT METATEX_ENABLE_LUA)
    return()
endif()
find_package(Lua 5.3 REQUIRED)
add_library(lmtex MODULE lmtex.c Bridge.cpp)
target_include_directories(lmtex PRIVATE ${LUA_INCLUDE_DIR})
target_link_libraries(lmtex PRIVATE metatex_core ${LUA_LIBRARIES})
set_target_properties(lmtex PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lua"
    INSTALL_RPATH "$ORIGIN/../../metatex")
install(TARGETS lmtex LIBRARY DESTINATION "lib/lua/${LUA_VERSION_MAJOR}.${LUA_VERSION_MINOR}")
install(FILES lmtex.h DESTINATION include/metatex/lua)
if(BUILD_TESTING AND METATEX_BUILD_TESTS)
    find_program(METATEX_LUA_EXECUTABLE NAMES "lua${LUA_VERSION_MAJOR}.${LUA_VERSION_MINOR}" "lua${LUA_VERSION_MAJOR}${LUA_VERSION_MINOR}" lua)
    if(METATEX_LUA_EXECUTABLE)
        add_test(NAME lua-binding COMMAND "${METATEX_LUA_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test.lua" "$<TARGET_FILE:lmtex>")
        set_tests_properties(lua-binding PROPERTIES TIMEOUT 30)
    endif()
endif()
