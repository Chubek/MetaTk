cmake_minimum_required(VERSION 3.18)
project(MetaTeX VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_BUILD_RPATH_USE_ORIGIN TRUE)
include(cmake/Dependencies.cmake)
option(METATEX_BUILD_TESTS "Build MetaTeX tests" ON)
set(METATEX_METAVECT_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../MetaVect" CACHE PATH "MetaVect source tree")
add_subdirectory("${METATEX_METAVECT_SOURCE_DIR}/scene" "metavect-scene")
add_library(metatex_core SHARED
    core/Lexer.cpp core/Engine.cpp core/IO.cpp core/Math.cpp core/Scripting.cpp core/External.cpp core/MetaML.cpp
    device/HTML.cpp device/PDF.cpp device/JSON.cpp device/CairoPango.cpp)
target_include_directories(metatex_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_include_directories(metatex_core SYSTEM PUBLIC $<BUILD_INTERFACE:${METATEX_DSLTK_DIR}>)
target_compile_features(metatex_core PUBLIC cxx_std_20)
target_link_libraries(metatex_core PUBLIC metavect_scene)
target_link_libraries(metatex_core PRIVATE metatex_metaml)
target_compile_options(metatex_core PRIVATE -Wall -Wextra -Wpedantic)
option(METATEX_ENABLE_PANGO "Use PangoCairo for font shaping and PDF" ON)
if(METATEX_ENABLE_PANGO)
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
        pkg_check_modules(PANGOCAIRO QUIET IMPORTED_TARGET pangocairo)
        if(PANGOCAIRO_FOUND)
            target_link_libraries(metatex_core PRIVATE PkgConfig::PANGOCAIRO)
            target_compile_definitions(metatex_core PRIVATE METATEX_HAVE_PANGOCAIRO=1)
            message(STATUS "MetaTeX: PangoCairo Unicode PDF renderer enabled")
        else()
            message(STATUS "MetaTeX: PangoCairo headers not found; using built-in basic PDF writer")
        endif()
    endif()
endif()
add_library(metatex_latex MODULE plugins/latex/LaTeX.cpp)
add_library(metatex_nekotex MODULE plugins/nekotex/NekoTeX.cpp)
add_library(metatex_metavect MODULE plugins/metavect/MetaVect.cpp)
foreach(plugin IN ITEMS metatex_latex metatex_nekotex metatex_metavect)
    set_target_properties(${plugin} PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR})
    target_link_libraries(${plugin} PRIVATE metatex_core)
endforeach()
add_executable(metatex driver/cli.cpp driver/Repl.cpp)
target_link_libraries(metatex PRIVATE metatex_core metatex_dynalo replxx::replxx)
target_compile_definitions(metatex PRIVATE METATEX_PLUGIN_SUFFIX="${CMAKE_SHARED_MODULE_SUFFIX}")
set_target_properties(metatex PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
    INSTALL_RPATH "$ORIGIN/../lib/metatex")
set_target_properties(metatex_core metatex_latex metatex_nekotex metatex_metavect PROPERTIES
    INSTALL_RPATH "$ORIGIN")
if(BUILD_TESTING AND METATEX_BUILD_TESTS)
    add_executable(metatex_cpp_api_tests tests/cpp_api.cpp)
    target_link_libraries(metatex_cpp_api_tests PRIVATE metatex_core)
    add_test(NAME cpp-api-dsl COMMAND metatex_cpp_api_tests "${CMAKE_CURRENT_BINARY_DIR}/cpp-api-fixtures")
    set_tests_properties(cpp-api-dsl PROPERTIES TIMEOUT 30)
    add_executable(metatex_cpp_example examples/native.cpp)
    target_link_libraries(metatex_cpp_example PRIVATE metatex_core)
    add_test(NAME cpp-native-example COMMAND metatex_cpp_example)
    add_executable(metatex_tests tests/tests.cpp)
    target_link_libraries(metatex_tests PRIVATE metatex_core)
    target_compile_options(metatex_tests PRIVATE -UNDEBUG)
    add_test(NAME core-tests COMMAND metatex_tests)
    add_executable(metatex_regressions tests/regressions.cpp)
    target_link_libraries(metatex_regressions PRIVATE metatex_core)
    add_test(NAME core-regressions COMMAND metatex_regressions "${CMAKE_CURRENT_BINARY_DIR}/core-fixtures")
    set_tests_properties(core-regressions PROPERTIES TIMEOUT 30)
    add_executable(metatex_vector_tests tests/vector.cpp plugins/metavect/MetaVect.cpp)
    target_link_libraries(metatex_vector_tests PRIVATE metatex_core)
    add_test(NAME vector-renderers COMMAND metatex_vector_tests "${CMAKE_CURRENT_BINARY_DIR}/vector-fixtures")
    add_executable(metatex_scripting_tests tests/scripting.cpp)
    target_link_libraries(metatex_scripting_tests PRIVATE metatex_core)
    add_test(NAME scripting-primitives COMMAND metatex_scripting_tests "${CMAKE_CURRENT_BINARY_DIR}/script-fixtures")
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME integration-python COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:metatex> ${CMAKE_CURRENT_SOURCE_DIR})
        add_test(NAME vector-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vector_integration.py $<TARGET_FILE:metatex>)
        add_test(NAME scripting-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/scripting_integration.py $<TARGET_FILE:metatex> ${CMAKE_CURRENT_SOURCE_DIR})
        set_tests_properties(scripting-integration PROPERTIES TIMEOUT 90)
        add_test(NAME replxx-terminal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/repl_terminal.py $<TARGET_FILE:metatex>)
        set_tests_properties(replxx-terminal PROPERTIES TIMEOUT 45)
    endif()
    add_test(NAME latex-html COMMAND metatex --format latex --target html -o ${CMAKE_CURRENT_BINARY_DIR}/example.html ${CMAKE_CURRENT_SOURCE_DIR}/examples/hello.ltx)
    add_test(NAME latex-pdf COMMAND metatex --format latex --target pdf -o ${CMAKE_CURRENT_BINARY_DIR}/example.pdf ${CMAKE_CURRENT_SOURCE_DIR}/examples/hello.ltx)
    add_test(NAME neko-html COMMAND metatex --format nekotex --target html -o ${CMAKE_CURRENT_BINARY_DIR}/neko.html ${CMAKE_CURRENT_SOURCE_DIR}/examples/modern.nltx)
endif()
install(TARGETS metatex RUNTIME DESTINATION bin)
foreach(launcher IN ITEMS mtex mlatex nlatex)
    configure_file("bin/${launcher}" "${CMAKE_CURRENT_BINARY_DIR}/${launcher}" COPYONLY)
endforeach()
install(PROGRAMS bin/mtex bin/mlatex bin/nlatex DESTINATION bin)
install(FILES "${METATEX_DYNALO_DIR}/LICENSE" DESTINATION share/doc/metatex/licenses RENAME dynalo-LICENSE)
install(FILES "${METATEX_REPLXX_DIR}/LICENSE.md" DESTINATION share/doc/metatex/licenses RENAME replxx-LICENSE)
if(EXISTS "${METATEX_HTTPLIB_INCLUDE_DIR}/LICENSE")
    install(FILES "${METATEX_HTTPLIB_INCLUDE_DIR}/LICENSE" DESTINATION share/doc/metatex/licenses RENAME cpp-httplib-LICENSE)
endif()
install(DIRECTORY "${METATEX_METAML_DIR}/metalib/" DESTINATION share/metaml/stdlib/cpp
    FILES_MATCHING PATTERN "*.ml" PATTERN "*.mml")
install(TARGETS metatex_core metatex_latex metatex_nekotex metatex_metavect LIBRARY DESTINATION lib/metatex)
install(TARGETS metavect_scene ARCHIVE DESTINATION lib/metatex)
install(FILES "${METATEX_METAVECT_SOURCE_DIR}/include/MetaVect-Scene.h" "${METATEX_METAVECT_SOURCE_DIR}/include/MetaVect-Scene.hpp" DESTINATION include)
install(DIRECTORY include/ DESTINATION include)
install(FILES "${METATEX_DSLTK_DIR}/DSLtk.hpp" DESTINATION include)
