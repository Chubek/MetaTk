cmake_minimum_required(VERSION 3.18)
project(MetaVMConsumer LANGUAGES C CXX)
enable_testing()
find_package(MetaVM 0.2 CONFIG REQUIRED COMPONENTS VMWeave Authoring Catalog Framework Machinery)
add_executable(c_consumer main.c generated.c)
set_target_properties(c_consumer PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_link_libraries(c_consumer PRIVATE MetaVM::VMWeave)
add_executable(cpp_consumer main.cpp)
set_target_properties(cpp_consumer PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(cpp_consumer PRIVATE MetaVM::Framework)
foreach(target c_consumer cpp_consumer)
 if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
 add_test(NAME ${target} COMMAND ${target})
endforeach()
