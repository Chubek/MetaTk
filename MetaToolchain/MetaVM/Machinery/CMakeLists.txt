cmake_minimum_required(VERSION 3.16)
project(MetaVMMachinery VERSION 0.1.0 LANGUAGES C)

option(METAVM_BUILD_TESTS "Build Machinery tests" ON)
option(METAVM_BUILD_EXAMPLES "Build the C11 example runner" ON)
option(METAVM_TEST_CPP "Build the optional C++11 wrapper test" OFF)
option(METAVM_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(METAVM_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" ON)

add_library(metavm
  MetaVM.c System.c Module.c Memory.c BinarySerialization.c Compiler.c Cache.c OS.c)
add_library(MetaVM::Machinery ALIAS metavm)
add_library(MetaVM::VM ALIAS metavm)
target_include_directories(metavm PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:include/MetaVM/Machinery>)
target_compile_features(metavm PUBLIC c_std_11)
set_target_properties(metavm PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES
  C_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON)

function(metavm_checks target)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(METAVM_WARNINGS_AS_ERRORS)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
    if(METAVM_SANITIZE)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()
metavm_checks(metavm)
if(METAVM_SANITIZE AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_link_options(metavm INTERFACE -fsanitize=address,undefined)
endif()

if(METAVM_BUILD_EXAMPLES)
  add_executable(metavm-run examples/run.c)
  target_link_libraries(metavm-run PRIVATE MetaVM::Machinery)
  set_target_properties(metavm-run PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
  metavm_checks(metavm-run)
endif()

if(METAVM_BUILD_TESTS)
  enable_testing()
  add_executable(metavm_test tests/test_machinery.c)
  target_link_libraries(metavm_test PRIVATE MetaVM::Machinery)
  set_target_properties(metavm_test PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
  metavm_checks(metavm_test)
  add_test(NAME metavm_machinery COMMAND metavm_test)
  if(METAVM_BUILD_EXAMPLES)
    add_test(NAME metavm_factorial COMMAND metavm-run "${CMAKE_CURRENT_SOURCE_DIR}/examples/factorial.mvm")
    add_test(NAME metavm_factorial_threaded COMMAND metavm-run "${CMAKE_CURRENT_SOURCE_DIR}/examples/factorial.mvm" --threaded)
    set_tests_properties(metavm_factorial metavm_factorial_threaded PROPERTIES PASS_REGULAR_EXPRESSION "3628800")
  endif()
  if(METAVM_TEST_CPP)
    enable_language(CXX)
    set_source_files_properties(tests/Cpp.hpp PROPERTIES LANGUAGE CXX)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
      set_source_files_properties(tests/Cpp.hpp PROPERTIES COMPILE_FLAGS "-x c++")
    elseif(MSVC)
      set_source_files_properties(tests/Cpp.hpp PROPERTIES COMPILE_FLAGS "/TP")
    endif()
    add_executable(metavm_test_cpp tests/Cpp.hpp)
    set_target_properties(metavm_test_cpp PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS OFF)
    target_link_libraries(metavm_test_cpp PRIVATE MetaVM::Machinery)
    metavm_checks(metavm_test_cpp)
    add_test(NAME metavm_cpp COMMAND metavm_test_cpp)
  endif()
endif()

include(GNUInstallDirs)
install(TARGETS metavm EXPORT MetaVMMachineryTargets
  ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
  LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
  RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
set_target_properties(metavm PROPERTIES EXPORT_NAME Machinery)
install(EXPORT MetaVMMachineryTargets NAMESPACE MetaVM::
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MetaVMMachinery)
install(FILES cmake/MetaVMMachineryConfig.cmake
  DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MetaVMMachinery)
file(GLOB public_headers "${CMAKE_CURRENT_SOURCE_DIR}/*.h" "${CMAKE_CURRENT_SOURCE_DIR}/*.hpp")
list(REMOVE_ITEM public_headers "${CMAKE_CURRENT_SOURCE_DIR}/Internal.h")
install(FILES ${public_headers} DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/MetaVM/Machinery)
