cmake_minimum_required(VERSION 3.16)
project(MetaVM VERSION 1.0.0 LANGUAGES CXX)
option(METAVM_BUILD_TESTS "Build register VM tests" ON)
option(METAVM_BUILD_WEAVE "Build the optional MetaIL-to-bytecode compiler" OFF)
add_library(metavm vmspec/VM.cpp)
add_library(MetaVM::VM ALIAS metavm)
target_include_directories(metavm PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(metavm PUBLIC cxx_std_17)
set_target_properties(metavm PROPERTIES CXX_EXTENSIONS OFF)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(metavm PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(METAVM_BUILD_TESTS)
  enable_testing()
  add_executable(metavm_test tests/vm.cpp)
  target_link_libraries(metavm_test PRIVATE MetaVM::VM)
  add_test(NAME metavm_runtime COMMAND metavm_test)
endif()
if(METAVM_BUILD_WEAVE)
  enable_language(C)
  if(NOT TARGET MetaIL::MetaIL)
    find_package(MetaIL CONFIG QUIET)
    if(NOT TARGET MetaIL::MetaIL)
      set(METAVM_METAIL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../MetaIL" CACHE PATH "MetaIL source checkout")
      if(NOT EXISTS "${METAVM_METAIL_ROOT}/CMakeLists.txt")
        message(FATAL_ERROR "Install MetaIL or set METAVM_METAIL_ROOT to its source checkout")
      endif()
      set(METAIL_BUILD_EXAMPLES OFF)
      add_subdirectory("${METAVM_METAIL_ROOT}" metail)
    endif()
  endif()
  add_library(metavm_weave vmweave/Weave.cpp)
  add_library(MetaVM::Weave ALIAS metavm_weave)
  target_link_libraries(metavm_weave PUBLIC MetaVM::VM PRIVATE MetaIL::MetaIL)
  target_compile_features(metavm_weave PUBLIC cxx_std_17)
  set_target_properties(metavm_weave PROPERTIES CXX_EXTENSIONS OFF)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(metavm_weave PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(metavm_weave_cli tools/vmweave.cpp)
  set_target_properties(metavm_weave_cli PROPERTIES OUTPUT_NAME vmweave CXX_EXTENSIONS OFF)
  target_link_libraries(metavm_weave_cli PRIVATE MetaVM::Weave)
  if(METAVM_BUILD_TESTS)
    add_executable(metavm_test_weave tests/weave.cpp)
    target_link_libraries(metavm_test_weave PRIVATE MetaVM::Weave)
    add_test(NAME metavm_weave COMMAND metavm_test_weave)
    add_test(NAME metavm_cli COMMAND ${CMAKE_COMMAND}
      -DTOOL=$<TARGET_FILE:metavm_weave_cli> -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-test
      -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
  endif()
endif()
