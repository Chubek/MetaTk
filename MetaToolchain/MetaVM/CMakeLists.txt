cmake_minimum_required(VERSION 3.18)
project(MetaVM VERSION 0.2.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(METAVM_BUILD_TESTS "Build tests" ON)
option(METAVM_ENABLE_METAML "Build real MetaML VMWeave front end (needs DSLtk.hpp)" OFF)
option(METAVM_ENABLE_VMDESC "Build VMDesc (needs externally built metatex_core)" OFF)
option(METAVM_ENABLE_LIBDRM "Build KMS dumb-buffer backend" OFF)
set(METAVM_DSLTK_INCLUDE_DIR "" CACHE PATH "Directory containing DSLtk.hpp")
set(METAVM_METATEX_SOURCE_DIR "" CACHE PATH "Optional original MetaTeX source root")
if(METAVM_BUILD_TESTS)
 enable_testing()
endif()
add_subdirectory(Machinery)
add_library(vmweave vmweave/vmweave.c)
target_link_libraries(vmweave PUBLIC metavm)
target_include_directories(vmweave PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/vmweave>
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/MetaVM/vmweave>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/MetaVM>)
add_library(MetaVM::VMWeave ALIAS vmweave)
set_target_properties(vmweave PROPERTIES EXPORT_NAME VMWeave WINDOWS_EXPORT_ALL_SYMBOLS ON)
metavm_checks(vmweave)
add_executable(vmweave-cli vmweave/main.c)
set_target_properties(vmweave-cli PROPERTIES OUTPUT_NAME vmweave)
target_link_libraries(vmweave-cli PRIVATE vmweave)
file(GLOB_RECURSE CATALOG_C CONFIGURE_DEPENDS VMLibrary/*.c vmspecs/*.c)
file(GLOB_RECURSE CATALOG_CPP CONFIGURE_DEPENDS VMLibrary/*.cpp)
add_library(metavm_catalog ${CATALOG_C} ${CATALOG_CPP})
target_include_directories(metavm_catalog PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/MetaVM>)
add_library(MetaVM::Catalog ALIAS metavm_catalog)
set_target_properties(metavm_catalog PROPERTIES EXPORT_NAME Catalog WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_compile_features(metavm_catalog PUBLIC cxx_std_17)
target_link_libraries(metavm_catalog PUBLIC metavm)
metavm_checks(metavm_catalog)
add_library(metavm_authoring INTERFACE)
add_library(MetaVM::Authoring ALIAS metavm_authoring)
set_target_properties(metavm_authoring PROPERTIES EXPORT_NAME Authoring)
target_compile_features(metavm_authoring INTERFACE cxx_std_17)
target_link_libraries(metavm_authoring INTERFACE vmweave)
add_library(metavm_framework INTERFACE)
add_library(MetaVM::Framework ALIAS metavm_framework)
set_target_properties(metavm_framework PROPERTIES EXPORT_NAME Framework)
target_link_libraries(metavm_framework INTERFACE metavm_authoring metavm_catalog)
if(METAVM_ENABLE_LIBDRM)
 find_package(PkgConfig REQUIRED)
 pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
 target_compile_definitions(metavm_catalog PUBLIC METAVM_HAVE_LIBDRM=1)
 target_link_libraries(metavm_catalog PRIVATE PkgConfig::DRM)
endif()
if(METAVM_ENABLE_METAML)
 if(NOT EXISTS "${METAVM_DSLTK_INCLUDE_DIR}/DSLtk.hpp")
  message(FATAL_ERROR "MetaML.hpp requires the matching DSLtk.hpp; set METAVM_DSLTK_INCLUDE_DIR")
 endif()
 add_executable(vmweave-metaml vmweave/metaml_main.cpp)
 target_include_directories(vmweave-metaml PRIVATE third_party vmweave ${METAVM_DSLTK_INCLUDE_DIR})
 target_compile_features(vmweave-metaml PRIVATE cxx_std_20)
 target_link_libraries(vmweave-metaml PRIVATE metavm)
 find_package(Threads REQUIRED)
 target_link_libraries(vmweave-metaml PRIVATE Threads::Threads)
 metavm_checks(vmweave-metaml)
 if(METAVM_BUILD_TESTS)
  add_executable(metavm_metaml tests/metaml.cpp)
  target_include_directories(metavm_metaml PRIVATE ${METAVM_DSLTK_INCLUDE_DIR})
  target_compile_features(metavm_metaml PRIVATE cxx_std_20)
  target_link_libraries(metavm_metaml PRIVATE metavm_catalog Threads::Threads)
  metavm_checks(metavm_metaml)
  add_test(NAME metavm_metaml COMMAND metavm_metaml ${CMAKE_CURRENT_SOURCE_DIR})
  set_tests_properties(metavm_metaml PROPERTIES TIMEOUT 60)
 endif()
endif()
if(METAVM_ENABLE_VMDESC)
 if(NOT TARGET metatex_core)
  if(NOT EXISTS "${METAVM_METATEX_SOURCE_DIR}/CMakeLists.txt")
   message(FATAL_ERROR "VMDesc requires metatex_core or METAVM_METATEX_SOURCE_DIR and MetaTeX's external dependencies")
  endif()
  add_subdirectory("${METAVM_METATEX_SOURCE_DIR}" metatex EXCLUDE_FROM_ALL)
 endif()
 add_executable(vmdesc VMDesc/main.cpp)
 target_compile_features(vmdesc PRIVATE cxx_std_20)
 target_link_libraries(vmdesc PRIVATE metatex_core)
endif()
if(METAVM_BUILD_TESTS)
 find_package(Python3 COMPONENTS Interpreter QUIET)
 if(Python3_FOUND)
  add_test(NAME metavm_catalog_metadata COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/validate_catalog.py)
 endif()
 add_executable(metavm_integration tests/integration.cpp)
 target_link_libraries(metavm_integration PRIVATE metavm_catalog vmweave)
 metavm_checks(metavm_integration)
 add_test(NAME metavm_integration COMMAND metavm_integration)
endif()

if(METAVM_BUILD_TESTS)
 add_executable(metavm_refinement tests/refinement.cpp)

 metavm_checks(metavm_refinement)

# Install the framework using the same directory layout as its relative public
# includes. The existing standalone Machinery package remains available.
install(TARGETS vmweave metavm_catalog metavm_authoring metavm_framework EXPORT MetaVMTargets
 ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
 LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(EXPORT MetaVMTargets NAMESPACE MetaVM:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MetaVM)
configure_package_config_file(cmake/MetaVMConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MetaVMConfig.cmake
 INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MetaVM)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MetaVMConfigVersion.cmake
 VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MetaVMConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MetaVMConfigVersion.cmake
 DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MetaVM)
install(FILES MetaVM.h MetaVM.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/MetaVM)
install(DIRECTORY vmweave VMLibrary vmspecs VMDesc DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/MetaVM
 FILES_MATCHING PATTERN "*.h" PATTERN "*.hpp")
install(FILES third_party/MetaML.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/MetaVM/third_party)
install(FILES third_party/DEPENDENCIES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/MetaVM/third_party)
install(FILES MANIFEST.yaml DESTINATION ${CMAKE_INSTALL_DATADIR}/MetaVM)
install(DIRECTORY VMLibrary/ DESTINATION ${CMAKE_INSTALL_DATADIR}/MetaVM/VMLibrary
 FILES_MATCHING PATTERN "*.mvm" PATTERN "*.vmweave" PATTERN "*.vmspc" PATTERN "*.vmdesc" PATTERN "*.ssa")
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DATADIR}/MetaVM/docs FILES_MATCHING PATTERN "*.md" PATTERN "*.json")
set(METAVM_INSTALL_TOOLS vmweave-cli ssavm-run vmweave-definition-demo closure-vm-demo)
foreach(tool metavm-run vmweave-metaml vmdesc)
 if(TARGET ${tool})
  list(APPEND METAVM_INSTALL_TOOLS ${tool})
 endif()
endforeach()
install(TARGETS ${METAVM_INSTALL_TOOLS} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# Relative install directories make shared-library tools relocatable as well.
if(UNIX)
 if(APPLE)
  set(METAVM_ORIGIN "@loader_path")
 else()
  set(METAVM_ORIGIN "$ORIGIN")
 endif()
 file(RELATIVE_PATH METAVM_BIN_TO_LIB "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_BINDIR}" "${CMAKE_INSTALL_PREFIX}/${CMAKE_INSTALL_LIBDIR}")
 set_target_properties(metavm vmweave metavm_catalog PROPERTIES INSTALL_RPATH "${METAVM_ORIGIN}")
 foreach(tool IN LISTS METAVM_INSTALL_TOOLS)
  set_target_properties(${tool} PROPERTIES INSTALL_RPATH "${METAVM_ORIGIN}/${METAVM_BIN_TO_LIB}")
 endforeach()
endif()
