cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(MetaPP VERSION 0.1.0 LANGUAGES C CXX)
  enable_testing()
endif()
option(METAPP_BUILD_TESTS "Build MetaPP tests" ON)
option(METAPP_BUILD_TOOLS "Build metacpp" ON)
add_library(metapp MetaPP.cpp io/lexer.cpp definition/define_macro.cpp
  expansion/macro_subst.cpp operations/arithmetic_condition.cpp
  directives/process.cpp directives/include.cpp)
add_library(MetaPP::metapp ALIAS metapp)
target_compile_features(metapp PUBLIC cxx_std_17)
set_target_properties(metapp PROPERTIES CXX_EXTENSIONS OFF WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(metapp PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:include/MetaPP> PRIVATE include)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(metapp PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(METAPP_BUILD_TOOLS)
  add_executable(metacpp driver/metacpp.c)
  target_compile_features(metacpp PRIVATE c_std_11)
  set_target_properties(metacpp PROPERTIES C_EXTENSIONS OFF)
  target_link_libraries(metacpp PRIVATE metapp)
  install(TARGETS metacpp RUNTIME DESTINATION bin)
endif()
if(METAPP_BUILD_TESTS)
  enable_testing()
  add_executable(metapp_tests tests/preprocessor.cpp)
  target_link_libraries(metapp_tests PRIVATE metapp)
  target_include_directories(metapp_tests PRIVATE include)
  target_compile_definitions(metapp_tests PRIVATE FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures")
  add_test(NAME metapp.unit COMMAND metapp_tests)
  add_executable(metapp_c_api tests/c_api.c)
  target_link_libraries(metapp_c_api PRIVATE metapp)
  add_test(NAME metapp.c_api COMMAND metapp_c_api)
  if(METAPP_BUILD_TOOLS)
    add_test(NAME metapp.cli COMMAND ${CMAKE_COMMAND}
      -DTOOL=$<TARGET_FILE:metacpp> -DFIXTURES=${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures
      -DWORK=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
    find_package(Python3 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
      add_test(NAME metapp.robustness COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/robustness.py $<TARGET_FILE:metacpp>)
    endif()
    if(Python3_Interpreter_FOUND AND CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
      add_test(NAME metapp.differential COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/tests/differential.py $<TARGET_FILE:metacpp> ${CMAKE_C_COMPILER})
    endif()
  endif()
endif()
install(TARGETS metapp EXPORT MetaPPTargets ARCHIVE DESTINATION lib LIBRARY DESTINATION lib RUNTIME DESTINATION bin)
install(FILES MetaPP.h MetaPP.hpp DESTINATION include/MetaPP)
install(EXPORT MetaPPTargets NAMESPACE MetaPP:: DESTINATION lib/cmake/MetaPP)
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/MetaPPConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MetaPPConfig.cmake INSTALL_DESTINATION lib/cmake/MetaPP)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MetaPPConfigVersion.cmake VERSION 0.1.0 COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MetaPPConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MetaPPConfigVersion.cmake DESTINATION lib/cmake/MetaPP)
