option(ARCHAS_LUA "Enable Lua templates and Lua directives" ON)
option(ARCHAS_PLUGINS "Enable shared-library directive plugins" ${UNIX})
set(ARCHAS_LUA_DIR "" CACHE PATH "Lua source directory (lua.h and lapi.c); overrides system Lua")

add_library(archas src/Frontend.cpp src/Expressions.cpp src/Lua.cpp src/Plugins.cpp src/Backend.cpp)
add_library(MetaArch::as ALIAS archas)
target_link_libraries(archas PUBLIC MetaArch::ir)
target_compile_features(archas PUBLIC cxx_std_20)
set_target_properties(archas PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
target_compile_definitions(archas PRIVATE
  ARCHAS_SOURCE_DATA="${PROJECT_SOURCE_DIR}"
  ARCHAS_INSTALL_DATA="${CMAKE_INSTALL_FULL_DATADIR}/metaarch"
  ARCHAS_LUA=$<BOOL:${ARCHAS_LUA}>
  ARCHAS_PLUGINS=$<BOOL:${ARCHAS_PLUGINS}>)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(archas PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ARCHAS_PLUGINS)
  if(NOT UNIX)
    message(FATAL_ERROR "ArchAS shared-library plugins currently require POSIX dlopen")
  endif()
  target_link_libraries(archas PRIVATE ${CMAKE_DL_LIBS})
endif()

if(ARCHAS_LUA)
  if(NOT ARCHAS_LUA_DIR)
    find_package(Lua 5.3 QUIET)
    if(NOT LUA_FOUND)
      foreach(_lua_dir
          "${PROJECT_SOURCE_DIR}/../third_party/lua"
          "${PROJECT_SOURCE_DIR}/../third_party/jiterati/third_party/lua"
          "${PROJECT_SOURCE_DIR}/../third_party/QaMRpp/third_party/Jiterati/third_party/lua")
        if(EXISTS "${_lua_dir}/lua.h")
          set(ARCHAS_LUA_DIR "${_lua_dir}")
          break()
        endif()
      endforeach()
    endif()
  endif()
  if(NOT ARCHAS_LUA_DIR AND LUA_FOUND)
    target_include_directories(archas PRIVATE ${LUA_INCLUDE_DIR})
    target_link_libraries(archas PRIVATE ${LUA_LIBRARIES})
  else()
    if(NOT ARCHAS_LUA_DIR)
      if(NOT ARCHDB_FETCH_DEPS)
        message(FATAL_ERROR "Lua >= 5.3 not found; set ARCHAS_LUA_DIR or ARCHAS_LUA=OFF")
      endif()
      FetchContent_Declare(archas_lua_source
        GIT_REPOSITORY https://github.com/lua/lua.git GIT_TAG v5.4.8 GIT_SHALLOW TRUE)
      FetchContent_MakeAvailable(archas_lua_source)
      set(ARCHAS_LUA_DIR "${archas_lua_source_SOURCE_DIR}")
    endif()
    if(EXISTS "${ARCHAS_LUA_DIR}/src/lua.h")
      set(ARCHAS_LUA_DIR "${ARCHAS_LUA_DIR}/src")
    endif()
    if(NOT EXISTS "${ARCHAS_LUA_DIR}/lua.h")
      message(FATAL_ERROR "ARCHAS_LUA_DIR must contain Lua sources")
    endif()
    file(GLOB _lua_sources "${ARCHAS_LUA_DIR}/*.c")
    list(FILTER _lua_sources EXCLUDE REGEX "/(lua|luac|onelua|ltests)\\.c$")
    add_library(archas_lua STATIC ${_lua_sources})
    set_target_properties(archas_lua PROPERTIES POSITION_INDEPENDENT_CODE ON)
    # Lua's portable switch dispatcher keeps strict C11/-Wpedantic builds clean.
    target_compile_definitions(archas_lua PRIVATE LUA_USE_JUMPTABLE=0)
    target_include_directories(archas_lua PUBLIC "${ARCHAS_LUA_DIR}")
    if(UNIX)
      target_compile_definitions(archas_lua PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)
      target_link_libraries(archas_lua PRIVATE ${CMAKE_DL_LIBS} m)
    endif()
    target_link_libraries(archas PRIVATE archas_lua)
    install(TARGETS archas_lua ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  endif()
endif()

  set_target_properties(domas PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
