cmake_minimum_required(VERSION 3.16)
project(MetaIL LANGUAGES C CXX)
include(GNUInstallDirs)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(METANUM_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../MetaNum" CACHE PATH "MetaNum header directory")
if(NOT EXISTS "${METANUM_ROOT}/all.hpp")
  message(FATAL_ERROR "MetaNum headers required; set METANUM_ROOT")
endif()
set(DSLTK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../DSLtk" CACHE PATH "DSLtk header directory")
if(NOT EXISTS "${DSLTK_ROOT}/DSLtk.hpp")
  message(FATAL_ERROR "DSLtk.hpp required for the C++ native DSL; set DSLTK_ROOT")
endif()
set(WEIGHSTONE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../weighstone" CACHE PATH "Weighstone source directory")
if(NOT EXISTS "${WEIGHSTONE_ROOT}/include/Weighstone/CCINT/KnownBits.hpp")
  message(FATAL_ERROR "Weighstone headers required; set WEIGHSTONE_ROOT")
endif()
find_package(Boost REQUIRED CONFIG)
# The Lua interpreter is provided by the official Lua 5.4 shared library.
# We ship API declarations for these narrow entry points; upstream headers are not copied.
find_library(LUA_54_LIBRARY NAMES lua5.4 lua5.4.so.0 liblua5.4.so.0
  PATHS /usr/lib/x86_64-linux-gnu /lib/x86_64-linux-gnu /usr/local/lib)
if(NOT LUA_54_LIBRARY)
  message(FATAL_ERROR "Lua 5.4 runtime library required: install liblua5.4-dev")
endif()
add_library(metanum_binding OBJECT lua/metanum.cpp)
set_target_properties(metanum_binding PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(metanum_binding PRIVATE "${METANUM_ROOT}")
add_library(metanum_native MODULE $<TARGET_OBJECTS:metanum_binding>)
set_target_properties(metanum_native PROPERTIES PREFIX "")
target_link_libraries(metanum_native PRIVATE ${LUA_54_LIBRARY})
add_library(weighstone_binding OBJECT lua/weighstone.cpp)
set_target_properties(weighstone_binding PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(weighstone_binding SYSTEM PRIVATE "${WEIGHSTONE_ROOT}/include" ${Boost_INCLUDE_DIRS})
add_library(weighstone_native MODULE $<TARGET_OBJECTS:weighstone_binding>)
set_target_properties(weighstone_native PROPERTIES PREFIX "")
target_link_libraries(weighstone_native PRIVATE ${LUA_54_LIBRARY})
add_library(metail_core MetaIL.c lua/lua.c lua/lkopt.c $<TARGET_OBJECTS:metanum_binding> $<TARGET_OBJECTS:weighstone_binding>)
add_library(MetaIL::MetaIL ALIAS metail_core)
set_target_properties(metail_core PROPERTIES OUTPUT_NAME MetaIL LINKER_LANGUAGE CXX)
target_compile_definitions(metail_core PRIVATE METAIL_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}"
  METAIL_INSTALL_ROOT="${CMAKE_INSTALL_FULL_DATADIR}/metail")
target_include_directories(metail_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
  $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/metail>)
target_include_directories(metail_core SYSTEM INTERFACE $<BUILD_INTERFACE:${DSLTK_ROOT}>)
target_compile_features(metail_core PUBLIC c_std_11 INTERFACE cxx_std_20)
target_link_libraries(metail_core PRIVATE ${LUA_54_LIBRARY} ${CMAKE_DL_LIBS} m)
add_executable(metail cli/metail.c)
target_link_libraries(metail PRIVATE MetaIL::MetaIL)
set_target_properties(metail PROPERTIES LINKER_LANGUAGE CXX)
# Keep the CLI usable when a shared-library installation is relocated.
file(RELATIVE_PATH METAIL_BIN_TO_LIB "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
if(APPLE)
  set_target_properties(metail PROPERTIES INSTALL_RPATH "@loader_path/${METAIL_BIN_TO_LIB}")
elseif(UNIX)
  set_target_properties(metail PROPERTIES INSTALL_RPATH "$ORIGIN/${METAIL_BIN_TO_LIB}")
endif()
if(MSVC)
  target_compile_options(metail PRIVATE /W4)
  target_compile_options(metail_core PRIVATE /W4)
  target_compile_options(metanum_binding PRIVATE /W4)
  target_compile_options(weighstone_binding PRIVATE /W4)
else()
  target_compile_options(metail PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(metail_core PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(metanum_binding PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(weighstone_binding PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(metail_test_structured tests/test_structured.c)
target_link_libraries(metail_test_structured PRIVATE MetaIL::MetaIL)

add_test(NAME dsl-rewrite COMMAND metail_test_dsl_rewrite ${CMAKE_CURRENT_SOURCE_DIR})

  $<TARGET_FILE:metail_test_dsl_rewrite> $<TARGET_FILE:metail>)
add_test(NAME integration COMMAND ${CMAKE_CURRENT_SOURCE_DIR}/tests/run.sh $<TARGET_FILE:metail>)

find_program(LUA_54_EXECUTABLE NAMES lua5.4)
if(LUA_54_EXECUTABLE)
  add_test(NAME standalone-lua COMMAND ${LUA_54_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/standalone.lua
    ${CMAKE_CURRENT_SOURCE_DIR} $<TARGET_FILE_DIR:metanum_native>)
  # The system Lua executable is not instrumented. Load the matching ASan runtime
  # before it dlopens an instrumented module (the embedded host already links it).
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_FLAGS MATCHES "-fsanitize=[^ ]*address")
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      set(asan_library_name libasan.so)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      set(asan_library_name "libclang_rt.asan-${CMAKE_SYSTEM_PROCESSOR}.so")
    endif()
    if(asan_library_name)
      execute_process(COMMAND ${CMAKE_CXX_COMPILER} "-print-file-name=${asan_library_name}"
        OUTPUT_VARIABLE asan_library OUTPUT_STRIP_TRAILING_WHITESPACE)
      if(EXISTS "${asan_library}")
        set_tests_properties(standalone-lua PROPERTIES ENVIRONMENT "LD_PRELOAD=${asan_library}")
      endif()
    endif()
  endif()
endif()
