cmake_minimum_required(VERSION 3.16)
project(MetaJIT VERSION 0.1.0 LANGUAGES C CXX)

option(METAJIT_BUILD_TESTS "Build MetaJIT tests and generated example" ON)
set(METAJIT_GPPX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../GPPX" CACHE PATH "GPPX source checkout")
set(METAJIT_METAIL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../MetaIL" CACHE PATH "MetaIL source checkout")

if(NOT TARGET GPPX::gppx)
  find_package(GPPX CONFIG QUIET)
  if(NOT TARGET GPPX::gppx)
    if(NOT EXISTS "${METAJIT_GPPX_ROOT}/CMakeLists.txt")
      message(FATAL_ERROR "Install GPPX or set METAJIT_GPPX_ROOT to its source checkout")
    endif()
    set(GPPX_BUILD_TESTS OFF)
    set(GPPX_BUILD_WEBBER OFF)
    set(GPPX_BUILD_EXAMPLES OFF)
    add_subdirectory("${METAJIT_GPPX_ROOT}" gppx EXCLUDE_FROM_ALL)
  endif()
endif()
if(NOT TARGET MetaIL::MetaIL)
  find_package(MetaIL CONFIG QUIET)
  if(NOT TARGET MetaIL::MetaIL)
    set(METAIL_BUILD_EXAMPLES OFF)
    add_subdirectory("${METAJIT_METAIL_ROOT}" metail)
  endif()
endif()

add_library(metajit_runtime Runtime.cpp)
add_library(MetaJIT::Runtime ALIAS metajit_runtime)
target_include_directories(metajit_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(metajit_runtime PUBLIC cxx_std_17)
set_target_properties(metajit_runtime PROPERTIES CXX_EXTENSIONS OFF)

add_library(metajit_preprocess Preprocess.cpp)
add_library(MetaJIT::Preprocess ALIAS metajit_preprocess)
target_include_directories(metajit_preprocess PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_features(metajit_preprocess PUBLIC cxx_std_17)
target_link_libraries(metajit_preprocess PRIVATE MetaIL::MetaIL GPPX::gppx)
set_target_properties(metajit_preprocess PROPERTIES CXX_EXTENSIONS OFF)

add_executable(metajit-preprocess tools/Preprocess.cpp)
target_link_libraries(metajit-preprocess PRIVATE MetaJIT::Preprocess)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  foreach(target metajit_runtime metajit_preprocess metajit-preprocess)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endforeach()
endif()

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/MetaJITPreprocess.cmake")
if(METAJIT_BUILD_TESTS)
  enable_testing()
  add_executable(metajit_test_preprocess tests/preprocess.cpp)
  target_link_libraries(metajit_test_preprocess PRIVATE MetaJIT::Preprocess)
  add_test(NAME metajit_preprocess COMMAND metajit_test_preprocess)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    metajit_preprocess_source(generated_example examples/affine.mjit)
    add_executable(metajit_example "${generated_example}")
    target_link_libraries(metajit_example PRIVATE MetaJIT::Runtime)
    add_test(NAME metajit_example COMMAND metajit_example)
    set_tests_properties(metajit_example PROPERTIES PASS_REGULAR_EXPRESSION "42")
    metajit_preprocess_source(generated_test tests/native.mjit DEPENDS tests/ops.mi)
    add_executable(metajit_test_native "${generated_test}")
    target_link_libraries(metajit_test_native PRIVATE MetaJIT::Runtime)
    add_test(NAME metajit_native COMMAND metajit_test_native)
  endif()
  add_test(NAME metajit_cli COMMAND ${CMAKE_COMMAND}
    -DTOOL=$<TARGET_FILE:metajit-preprocess>
    -DWORK=${CMAKE_CURRENT_BINARY_DIR}/cli-test
    -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.cmake)
endif()
