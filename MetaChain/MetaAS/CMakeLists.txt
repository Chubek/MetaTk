cmake_minimum_required(VERSION 3.24)
project(MetaAS VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
option(METAAS_BUILD_TOOLS "Build the MetaAS command-line assembler" ON)
option(METAAS_WARNINGS_AS_ERRORS "Treat MetaAS warnings as errors" ON)
option(METAAS_SANITIZERS "Instrument MetaAS and its MetaArch dependency with ASan/UBSan" OFF)
set(METAAS_METAARCH_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../MetaArch" CACHE PATH "MetaArch source directory")

if(METAAS_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
if(NOT TARGET MetaArch::dsl)
    if(NOT EXISTS "${METAAS_METAARCH_DIR}/MetaArch.hpp")
        message(FATAL_ERROR "MetaArch not found; set METAAS_METAARCH_DIR")
    endif()
    set(ARCHDB_BUILD_DSL ON)
    set(ARCHDB_BUILD_TESTS OFF)
    set(ARCHDB_BUILD_TOOLS OFF)
    add_subdirectory("${METAAS_METAARCH_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/metaarch")
    if(UNIX)
        if(APPLE)
            set(_metaas_loader "@loader_path")
        else()
            set(_metaas_loader "$ORIGIN")
        endif()
        set_target_properties(metaarch metaarch_dsl PROPERTIES INSTALL_RPATH "${_metaas_loader}")
    endif()
endif()

add_library(metaas src/Assembler.cpp src/Image.cpp src/CAPI.cpp)
add_library(MetaAS::MetaAS ALIAS metaas)
target_compile_features(metaas PUBLIC cxx_std_20)
set_target_properties(metaas PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON WINDOWS_EXPORT_ALL_SYMBOLS ON)
target_include_directories(metaas PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(metaas PRIVATE MetaArch::dsl)
function(metaas_warnings target)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
        if(METAAS_WARNINGS_AS_ERRORS)
            target_compile_options(${target} PRIVATE -Werror)
        endif()
    endif()
endfunction()
metaas_warnings(metaas)
if(UNIX)
    if(APPLE)
        set(_metaas_loader "@loader_path")
    else()
        set(_metaas_loader "$ORIGIN")
    endif()
    set_target_properties(metaas PROPERTIES INSTALL_RPATH "${_metaas_loader}")
endif()
if(METAAS_BUILD_TOOLS)
    add_executable(metaas_cli src/Main.cpp)
    set_target_properties(metaas_cli PROPERTIES OUTPUT_NAME metaas CXX_EXTENSIONS OFF)
    target_link_libraries(metaas_cli PRIVATE MetaAS::MetaAS)
    metaas_warnings(metaas_cli)
    if(UNIX)
        file(RELATIVE_PATH _metaas_lib_from_bin "${CMAKE_INSTALL_FULL_BINDIR}" "${CMAKE_INSTALL_FULL_LIBDIR}")
        set_target_properties(metaas_cli PROPERTIES INSTALL_RPATH "${_metaas_loader}/${_metaas_lib_from_bin}")
    endif()
    install(TARGETS metaas_cli RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS metaas ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/MetaAS.h include/MetaAS.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})

if(BUILD_TESTING)
    add_executable(metaas_tests tests/Assembler.cpp)
    target_link_libraries(metaas_tests PRIVATE MetaAS::MetaAS)
    metaas_warnings(metaas_tests)
    add_test(NAME metaas.core COMMAND metaas_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture.march" "${METAAS_METAARCH_DIR}")
    add_executable(metaas_c_tests tests/CAPI.c)
    set_target_properties(metaas_c_tests PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    target_link_libraries(metaas_c_tests PRIVATE MetaAS::MetaAS)
    metaas_warnings(metaas_c_tests)
    add_test(NAME metaas.c_abi COMMAND metaas_c_tests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture.march")
    if(METAAS_BUILD_TOOLS)
        add_test(NAME metaas.help COMMAND metaas_cli --help)
        find_package(Python3 COMPONENTS Interpreter QUIET)
        if(Python3_Interpreter_FOUND)
            add_test(NAME metaas.integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py"
                "$<TARGET_FILE:metaas_cli>" "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixture.march" "${METAAS_METAARCH_DIR}")
        endif()
    endif()
endif()
